\begin{multicols}{2}

\subsection{Basics}
Docs: \url{https://docs.pytest.org/en/stable/}

\begin{eqlist}
	\item[\texttt{pytest}] Run all tests pytest can find
	\item[\texttt{pytest <file> <file>}] Run tests in specific files
\end{eqlist}

\subsection{Selecting Tests}

Example: \texttt{pytest tests/test_sim.py -k sff -m parallel}

\begin{eqlist}
	\item[\texttt{-k EXPR}] Only run tests matching given expression
	\item[\texttt{-m MARK}] Only run tests matching given mark
	\item[\texttt{-x, --exitfirst}] Exit on first error or failed test
	\item[\texttt{--maxfail=num}] Exit after first num failures or errors
	\item[\texttt{--lf}] Rerun only the tests that failed at last run
	\item[\texttt{--ff}] Run all tests, but run last failures first
	\item[\texttt{--nf}] Run tests from new files first and then tests sorted by file mtime
\end{eqlist}

\subsection{Inspecting Settings}
\begin{eqlist}
	\item[\texttt{--markers}] Show markers
	\item[\texttt{--fixtures}] Show available fixtures
	\item[\texttt{--durations=N}] Show N slowest setup/test durations (N=0 for all)
\end{eqlist}

\subsection{Run Settings}
\begin{eqlist}
	\item[\texttt{-v}] Increase verbosity
	\item[\texttt{-vv}] Even more verbose!
	\item[\texttt{-s}] Show stdout and stderr in real time
	\item[\texttt{--co}] Only collect tests, don't execute them
	\item[\texttt{--trace}] Immediately break when running each test
	\item[\texttt{-p name}] Early-load given plugin module name or entry point. To avoid loading of plugins, use the "no:" prefix, e.g., "no:doctest"
\end{eqlist}

% \columnbreak
\subsection{Parallel Execution: \texttt{pytest-xdist}}

\begin{eqlist}
	\item[\texttt{-n}] Set the number of processes to run
	\item[\texttt{--dist=distmode}] Set the distribution mode; "each", "load", "loadscope", "loadfile", "loadgroup", "worksteal"
\end{eqlist}
					
\subsection{Code Coverage: \texttt{pytest-cov}}

\begin{eqlist}
	\item[\texttt{--cov=[SOURCE]}] Path or package name to measure during execution (multi-allowed)
	\item[\texttt{--cov-branch}] Enable branch coverage
\end{eqlist}
